def _minimal_image(filename='test.jpg'):
    """Vrátí SimpleUploadedFile s minimálním platným JPEG (1×1 px) generovaným Pillow."""
    buf = io.BytesIO()
    PILImage.new('RGB', (1, 1), color=(255, 0, 0)).save(buf, format='JPEG')
    return SimpleUploadedFile(filename, buf.getvalue(), content_type='image/jpeg')

Ukázka testu (vytvoření článku)

def test_create_post_with_image(self):
    """Přihlášený uživatel může vytvořit článek s obrázkem."""
    self.client.credentials(HTTP_AUTHORIZATION=f'Bearer {self._get_token(self.user)}')
    payload = {
        'title':     'Test Drone Shot',
        'content':   'Popis fotky z dronu.',
        'status':    'published',
        'location':  'Brno, CZ',
        'altitude':  150,
        'drone_model': 'DJI Air 3',
        'cover_image': _minimal_image('cover.jpg'),
    }
    res = self.client.post('/api/posts/', payload, format='multipart')
    self.assertEqual(res.status_code, status.HTTP_201_CREATED)
    post = Post.objects.get(slug='test-drone-shot')
    self.assertEqual(post.altitude, 150)
    self.assertTrue(post.cover_image.name.endswith('.jpg'))

def test_create_post_unauthorized(self):
    """Nepřihlášený uživatel dostane 401."""
    res = self.client.post('/api/posts/', {'title': 'X', 'content': 'Y'})
    self.assertEqual(res.status_code, status.HTTP_401_UNAUTHORIZED)

🛡️ MSW Mock Server

MSW (Mock Service Worker) interceptuje HTTP požadavky na síťové vrstvě — bez úprav kódu komponent.

Handlers (test/mocks/handlers.js)

import { http, HttpResponse } from 'msw'
import { mockPostsPage, mockPost, mockTokens, mockUser } from './data'

export const handlers = [
  // Seznam článků
  http.get('/api/posts/', () => HttpResponse.json(mockPostsPage)),

  // Detail článku
  http.get('/api/posts/praha-z-vysky-120-metru/', () => HttpResponse.json(mockPost)),
  http.get('/api/posts/:slug/', ({ params }) =>
    params.slug === 'neexistujici-clanek'
      ? HttpResponse.json({ detail: 'Not found.' }, { status: 404 })
      : HttpResponse.json(mockPost)
  ),

  // Přihlášení
  http.post('/api/auth/login/', () => HttpResponse.json(mockTokens)),

  // Aktuální uživatel — POUZE s platným test tokenem
  http.get('/api/auth/me/', ({ request }) => {
    const auth = request.headers.get('Authorization')
    if (auth !== 'Bearer fake.access.token') {
      return HttpResponse.json({ detail: 'Unauthorized.' }, { status: 401 })
    }
    return HttpResponse.json(mockUser)
  }),

  // Kategorie a tagy
  http.get('/api/categories/', () => HttpResponse.json([mockCategory])),
  http.get('/api/tags/', () => HttpResponse.json([mockTag])),

  // Moje články
  http.get('/api/posts/my/', () => HttpResponse.json(mockPostsPage)),

  // Registrace
  http.post('/api/auth/register/', () =>

  ),
]

🧰 Test helpers

test/helpers.jsx

/**
 * renderWithProviders — obalí komponentu všemi potřebnými providery:
 * QueryClientProvider + MemoryRouter + AuthContext
 */
export function renderWithProviders(ui, { initialEntries = ['/'] } = {}) {
  const queryClient = new QueryClient({
    defaultOptions: { queries: { retry: false } }
  })
  return render(
    <QueryClientProvider client={queryClient}>
      <MemoryRouter initialEntries={initialEntries}>
        <AuthProvider>{ui}</AuthProvider>
      </MemoryRouter>
    </QueryClientProvider>
  )
}

/** Nastaví localStorage jako by byl uživatel přihlášen */
export function mockLoggedIn(token = 'fake.access.token') {
  localStorage.setItem('access', token)
  localStorage.setItem('refresh', 'fake.refresh.token')
}

/** Vyčistí localStorage (odhlášení) */
export function mockLoggedOut() {
  localStorage.removeItem('access')
  localStorage.removeItem('refresh')
}

Proč MemoryRouter?

Komponenty jako PostDetail používají useParams() — k tomu musí být uvnitř Router kontextu. V testech používáme MemoryRouter s initialEntries (žádný skutečný prohlížeč URL).

PostDetail v testech

Komponenty používající useParams() musí být zabaleny v Routes + Route:

function PostDetailRoute() {
  return (
    <Routes>
      <Route path="/post/:slug" element={<PostDetail />} />
      <Route path="*" element={<div>not found page</div>} />
    </Routes>
  )
}

it('zobrazí titulek článku', async () => {
  renderWithProviders(<PostDetailRoute />, {
    initialEntries: ['/post/praha-z-vysky-120-metru'],
  })
  await waitFor(() => {
    expect(screen.getByRole('heading', {
      name: /Praha z výšky 120 metrů/
    })).toBeInTheDocument()
  }, { timeout: 3000 })
})
