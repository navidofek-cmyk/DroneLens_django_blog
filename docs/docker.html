Dockerfile.backend

FROM python:3.12-slim

# System deps (libpq pro psycopg2, gcc pro kompilaci)
RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq-dev gcc curl \
    && rm -rf /var/lib/apt/lists/*

# UV package manager
COPY --from=ghcr.io/astral-sh/uv:latest /uv /usr/local/bin/uv

# 🔒 Non-root user — bezpečnostní best practice
RUN groupadd -r appuser && useradd -r -g appuser appuser

WORKDIR /app

# Instalace závislostí (cache layer — pouze pokud se změní pyproject.toml)
COPY pyproject.toml uv.lock ./
RUN uv sync --frozen --no-dev

COPY . .

# Media a staticfiles adresáře s správnými právy
RUN mkdir -p media staticfiles \
    && chown -R appuser:appuser /app

COPY docker/backend-entrypoint.sh /entrypoint.sh
RUN chmod +x /entrypoint.sh

USER appuser   # 🔒 od teď neběžíme jako root
EXPOSE 8555
ENTRYPOINT ["/entrypoint.sh"]

Klíčové rozhodnutí

  • python:3.12-slim: minimální image bez zbytečných nástrojů
  • UV místo pip: 10–100× rychlejší instalace závislostí
  • --frozen: instaluje přesně verze z uv.lock — reprodukovatelné buildy
  • --no-dev: dev závislosti (pytest, ruff) se do produkčního image neinstalují
  • Non-root user: kontejner neběží jako root → menší bezpečnostní riziko
  • Layers cache: pyproject.toml + uv.lock se kopírují před kódem → cache se invaliduje jen při změně deps

Dockerfile.frontend

# ─── Dev target — Vite HMR ─────────────────────────────────────────────────
FROM node:22-alpine AS dev
WORKDIR /app
COPY frontend/package*.json ./
RUN npm ci
COPY frontend/ .
EXPOSE 5173
CMD ["npm", "run", "dev", "--", "--host"]

# ─── Build stage ────────────────────────────────────────────────────────────
FROM node:22-alpine AS builder
WORKDIR /app
COPY frontend/package*.json ./
RUN npm ci
COPY frontend/ .
RUN npm run build

# ─── Production — Nginx obsluhuje React build ──────────────────────────────
FROM nginx:alpine AS prod
COPY --from=builder /app/dist /usr/share/nginx/html
COPY docker/nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 80

Multi-stage build

  • dev: spouští Vite dev server s HMR (hot module replacement) — kód se aktualizuje bez restartu
  • builder: vytvoří produkční build (npm run build → dist/)
  • prod: minimální Nginx image, kopíruje pouze finální HTML/CSS/JS z builderu

docker-compose.yml (development)

services:

  db:
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_DB: droneview
      POSTGRES_USER: droneuser
      POSTGRES_PASSWORD: dronepass
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U droneuser -d droneview"]
      interval: 5s
      retries: 10

  backend:
    build:
      context: .
      dockerfile: Dockerfile.backend
    depends_on:
      db:
        condition: service_healthy   # čeká na zdravou DB
    environment:
      DJANGO_SECRET_KEY: "super-secret-change-in-production-xyz-123"
      DJANGO_DEBUG: "True"
      DATABASE_URL: "postgresql://droneuser:dronepass@db:5432/droneview"
      CORS_ALLOWED_ORIGINS: "http://localhost:5173"
    volumes:
      - media_data:/app/media
      - .:/app           # live reload — změny kódu viditelné okamžitě
    ports:
      - "8555:8555"

  frontend:
    build:
      context: .
      dockerfile: Dockerfile.frontend
      target: dev        # použije pouze dev stage
    depends_on: [backend]
    volumes:
      - ./frontend:/app
      - /app/node_modules   # node_modules je izolováno v kontejneru
    ports:
      - "5173:5173"

docker-compose.prod.yml (production)

services:

  db:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: ${DB_NAME}
      POSTGRES_USER: ${DB_USER}
      POSTGRES_PASSWORD: ${DB_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data

  backend:
    image: ghcr.io/your-org/django_blog-backend:${TAG:-latest}
    environment:
      DJANGO_SECRET_KEY: ${DJANGO_SECRET_KEY}
      DJANGO_DEBUG: "False"
      DJANGO_ALLOWED_HOSTS: ${DJANGO_ALLOWED_HOSTS}
      DATABASE_URL: postgresql://${DB_USER}:${DB_PASSWORD}@db:5432/${DB_NAME}
      # S3 storage pro media
      USE_S3: "True"
      AWS_ACCESS_KEY_ID: ${AWS_ACCESS_KEY_ID}
      AWS_SECRET_ACCESS_KEY: ${AWS_SECRET_ACCESS_KEY}
      AWS_STORAGE_BUCKET_NAME: ${AWS_STORAGE_BUCKET_NAME}
    depends_on:
      db:
        condition: service_healthy

  frontend:
    image: ghcr.io/your-org/django_blog-frontend:${TAG:-latest}
    build:
      context: .
      dockerfile: Dockerfile.frontend
      target: prod       # produkční Nginx build
    ports:
      - "80:80"
    depends_on: [backend]

Nginx konfigurace

Soubor docker/nginx.conf — Nginx obsluhuje React SPA a přeposílá API požadavky na Django:

server {
    listen 80;
    root /usr/share/nginx/html;
    index index.html;

    # React SPA — neznámé URL → index.html (client-side routing)
    location / {
        try_files $uri $uri/ /index.html;
    }

    # API → Django backend (Gunicorn)
    location /api/ {
        proxy_pass       http://backend:8555;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # Media a admin → Django
    location /media/  { proxy_pass http://backend:8555; }
    location /admin/  { proxy_pass http://backend:8555; }
    location /static/ { proxy_pass http://backend:8555; }

    gzip on;
    gzip_types text/css application/javascript application/json image/svg+xml;
}

Proč try_files $uri $uri/ /index.html?

React Router pracuje na straně klienta. Nginx musí vracet index.html pro libovolnou URL (např. /post/muj-clanek), aby React mohl URL zpracovat. Bez tohoto nastavení by Nginx vrátil 404.

Backend entrypoint script

Soubor docker/backend-entrypoint.sh:

#!/bin/bash
set -e

# Počkej na databázi (zdravotní kontrola)
echo "Čekám na databázi..."
until uv run python manage.py check --database default 2>/dev/null; do
  sleep 1
done

# Automatická migrace
echo "Spouštím migrace..."
uv run python manage.py migrate --noinput

# Statické soubory
echo "Sbírám statické soubory..."
uv run python manage.py collectstatic --noinput

# Spuštění Gunicorn
echo "Startuji Gunicorn..."
exec uv run gunicorn django_blog.wsgi:application \
    --bind 0.0.0.0:8555 \
    --workers 4 \
    --timeout 120

Užitečné Docker příkazy

# Spuštění dev stacku
docker compose up --build

# Spuštění na pozadí
docker compose up -d

# Logy konkrétního kontejneru
docker compose logs -f backend

# Spuštění Django příkazů
docker compose exec backend uv run python manage.py migrate
docker compose exec backend uv run python manage.py createsuperuser
docker compose exec backend uv run python manage.py shell

# Resetování DB (VAROVÁNÍ: smaže data!)
docker compose down -v

# Produkční build
docker compose -f docker-compose.prod.yml up --build

# Rebuild pouze jednoho kontejneru
docker compose up --build backend
Dev vs Prod volumes V dev compose souboru je backend volume .:/app — celý zdrojový kód je mount do kontejneru. To umožňuje live reload, ale nesmí se použít v produkci. Produkce používá image z registru.
